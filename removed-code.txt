2026-08-24 - workflows/lib/run-step.ts (whole file, deleted - RESTORED 2026-08-24)

The switch-based step runner, replaced by the registry-keyed map in
workflows/nodes/node-executors.ts with one implementation file per node under
workflows/nodes/executors/. The logic carried over unchanged; only the dispatch
shape differs, and the browser is now lazy. The file was never committed, so this
is the only copy.

    import type { Stagehand } from "@browserbasehq/stagehand"

    import type { StepNodeType } from "@/workflows/nodes/settings"

    // Runs one node against the shared browser session and returns what it produced
    // as a short string, so the result drops straight into run metadata and the logs
    // panel. Throwing here fails the step, and the task decides what that means for
    // the rest of the run.
    export async function runStep(
      stagehand: Stagehand,
      node: StepNodeType
    ): Promise<string | undefined> {
      const { type, values } = node.data

      switch (type) {
        // The trigger only marks where the graph starts, so there is nothing to do.
        case "start":
          return undefined

        case "open-url": {
          const url = values.url?.trim()

          if (!url) {
            throw new Error("Open URL node needs a url")
          }

          const page =
            stagehand.context.pages()[0] ?? (await stagehand.context.newPage())

          // domcontentloaded rather than a fixed sleep: act() and extract() reason
          // over the DOM, so waiting for every image to land buys nothing.
          await page.goto(url, { waitUntil: "domcontentloaded" })

          return `Opened ${url}`
        }

        case "act": {
          const instruction = values.instruction?.trim()

          if (!instruction) {
            throw new Error("Act node needs an instruction")
          }

          // observe() first, then act on the action it returns. The action is what
          // the cache stores, so a repeat run replays it without paying for
          // inference again. A miss falls back to the plain instruction.
          const [action] = await stagehand.observe(instruction)
          const result = action
            ? await stagehand.act(action)
            : await stagehand.act(instruction)

          return result.message
        }

        case "extract": {
          const instruction = values.instruction?.trim()

          if (!instruction) {
            throw new Error("Extract node needs an instruction")
          }

          const { extraction } = await stagehand.extract(instruction)

          return extraction
        }

        case "agent": {
          const goal = values.goal?.trim()

          if (!goal) {
            throw new Error("Agent node needs a goal")
          }

          // No scripted steps: the agent decides them at runtime. This is the node
          // that spends the most Model Gateway tokens.
          const result = await stagehand.agent().execute(goal)

          if (!result.success) {
            throw new Error(result.message)
          }

          return result.message
        }
      }
    }

2026-08-24 - workflows/lib/browserbase.ts (replaced export)

withBrowserSession, replaced by createBrowser() in the same file. The old helper
opened a cloud browser for every run, including graphs whose nodes never touch
one; createBrowser starts it on the first getStagehand() call instead.

    // One cloud browser per workflow run: every step shares the session, so a page
    // one node opened is still open for the node after it. The Browserbase API key
    // alone identifies the project, and leaving model unset routes the LLM calls
    // through Model Gateway on that same key, so no provider key is needed.
    export async function withBrowserSession<T>(
      run: (stagehand: Stagehand) => Promise<T>
    ): Promise<T> {
      if (!process.env.BROWSERBASE_API_KEY) {
        throw new Error("BROWSERBASE_API_KEY is not set")
      }

      const stagehand = new Stagehand({
        env: "BROWSERBASE",
        apiKey: process.env.BROWSERBASE_API_KEY,
        verbose: 0,
        // A cached act() replays without an LLM call. The directory lives on the
        // machine running the task, so this pays off across steps and across local
        // dev runs rather than across deployed ones.
        cacheDir: ".stagehand-cache",
      })

      await stagehand.init()

      try {
        return await run(stagehand)
      } finally {
        await stagehand.close()
      }
    }
